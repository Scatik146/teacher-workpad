dezhurstvo.js
```js
/* Данные. График: кто в какой день дежурит.
   Образец записи: { kto: 'Тимур', den: 'понедельник' } */
const GRAFIK = [
  { kto: 'Тимур', den: 'понедельник' },
  { kto: 'Глеб', den: 'вторник' },
  { kto: 'Вера', den: 'среда' },
];

// дежурный: имя того, кто дежурит в этот день, или пустая строка
function dezhurnyj(den, grafik) {
  let imya = '';
  for (const zapis of grafik) {
    if (zapis.den === den) imya = zapis.kto;
  }
  return imya;
}

if (typeof module !== 'undefined') module.exports = { GRAFIK, dezhurnyj };
```

# ДРАКОН-лист: дежурство

## Действия: слово «дежурный»

```дракон
название: dezhurnyj(den, grafik)
ввод: den
imya = ''
для: каждой записи графика
  вопрос: день записи равен den
    да:
      imya = имя из записи
    нет:
вывод: imya
```

## Данные и правила

```конспект
тема: Дежурство: кто кого сменяет
режим: знания
А -> ?
Б -> ?
Д -> ?
Е -> ?
связь Тимур -> понедельник: дежурит
связь Глеб -> вторник: дежурит
связь Вера -> среда: дежурит
связь вторник -> понедельник: после
связь среда -> вторник: после
правило: А сменяет Б, если А дежурит Д и Б дежурит Е и Д после Е
пример: Глеб сменяет Тимура
контрпример: Тимур сменяет Глеба
запрос: кто сменяет Глеба
```
